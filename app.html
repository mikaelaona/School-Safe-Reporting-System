from flask import Flask, request, redirect, url_for import sqlite3 import random import string import os app = Flask(__name__) DATABASE = "schoolfix.db" # ========================= # DATABASE # ========================= def init_db(): conn = sqlite3.connect(DATABASE) conn.execute(""" CREATE TABLE IF NOT EXISTS reports ( id INTEGER PRIMARY KEY AUTOINCREMENT, report_id TEXT UNIQUE, facility TEXT, location TEXT, description TEXT, urgency TEXT, status TEXT ) """) conn.commit() conn.close() def generate_report_id(): code = ''.join( random.choices( string.ascii_uppercase + string.digits, k=6 ) ) return "SF-" + code # ========================= # DESIGN # ========================= STYLE = """ """ # ========================= # HOME PAGE # ========================= @app.route("/") def home(): return STYLE + """

🏫 SchoolFix

School Facility Reporting System

Help Improve Our School

Found a broken or unsafe school facility? Report it through SchoolFix so the school administration can take action.

📝 Report a Problem 🔎 Check Report Status 👨‍💼 Admin Dashboard
SchoolFix © 2026
""" # ========================= # REPORT PAGE # ========================= @app.route("/report", methods=["GET", "POST"]) def report(): if request.method == "POST": facility = request.form["facility"] location = request.form["location"] description = request.form["description"] urgency = request.form["urgency"] report_id = generate_report_id() conn = sqlite3.connect(DATABASE) conn.execute(""" INSERT INTO reports ( report_id, facility, location, description, urgency, status ) VALUES (?, ?, ?, ?, ?, ?) """, ( report_id, facility, location, description, urgency, "Submitted" )) conn.commit() conn.close() return STYLE + f"""

🏫 SchoolFix

✅ Report Submitted!

Your facility report has been successfully submitted.

Your Report ID is:

{report_id}

Please save your Report ID so you can check the status of your report.

🔎 Check Status 🏠 Home
""" return STYLE + """

📝 Report a Problem

Facility Report Form


← Back to Home
""" # ========================= # STATUS PAGE # ========================= @app.route("/status", methods=["GET", "POST"]) def status(): result = "" if request.method == "POST": report_id = request.form["report_id"] conn = sqlite3.connect(DATABASE) report = conn.execute(""" SELECT * FROM reports WHERE report_id = ? """, (report_id,)).fetchone() conn.close() if report: result = f"""

Report Found ✅

Report ID: {report[1]}

Facility: {report[2]}

Location: {report[3]}

Urgency: {report[5]}

Status: {report[6]}

""" else: result = """

❌ Report ID not found.

""" return STYLE + f"""

🔎 Check Report Status

Track Your Report

{result}
← Back to Home
""" # ========================= # ADMIN DASHBOARD # ========================= @app.route("/admin") def admin(): conn = sqlite3.connect(DATABASE) reports = conn.execute(""" SELECT * FROM reports ORDER BY id DESC """).fetchall() conn.close() rows = "" for report in reports: rows += f""" {report[1]} {report[2]} {report[3]} {report[4]} {report[5]} {report[6]}
""" return STYLE + f"""

👨‍💼 Admin Dashboard

Facility Reports

{rows}
Report ID Facility Location Description Urgency Status Action

← Back to Home
""" # ========================= # UPDATE STATUS # ========================= @app.route("/update/", methods=["POST"]) def update(id): status = request.form["status"] conn = sqlite3.connect(DATABASE) conn.execute(""" UPDATE reports SET status = ? WHERE id = ? """, (status, id)) conn.commit() conn.close() return redirect(url_for("admin")) # ========================= # RUN APPLICATION # ========================= init_db() if __name__ == "__main__": port = int( os.environ.get("PORT", 5000) ) app.run( host="0.0.0.0", port=port, debug=True )
